Analyze Job Demo (client-side polling)

After POST /analyze or POST /analyze/save you receive: {"job_id": "..."}

Use the snippet below to poll job status and show results.

// Example JS (fetch API)
async function pollJob(jobId){
  const url = `/jobs/${jobId}`;
  while(true){
    const r = await fetch(url);
    const json = await r.json();
    console.log('status', json.status);
    if(json.status === 'completed'){
      // render json.result
      document.getElementById('out').textContent = JSON.stringify(json.result, null, 2);
      break;
    }
    if(json.status === 'failed'){
      alert('Job failed: '+json.error);
      break;
    }
    // show spinner/progress
    document.getElementById('status').innerHTML = ' Waiting...';
    await new Promise(res=>setTimeout(res, 1500));
  }
}
